use nepl3_core::{budget::{Budget,Limits,StopReason},schema::{SchemaRegistry,TypeDescriptor},value::NdfValue,value_codec::FoundationCodecError};
fn main(){
 let mut registry=SchemaRegistry::default();let mut setup=Budget::new(Limits{source_bytes:10000,work:100000,depth:100,nodes:10000,allocation_units:100000,output_bytes:100000,diagnostics:10,events:10});
 registry.finalize(&mut setup).unwrap();
 let mut limits=setup.limits();limits.work=0;let mut budget=Budget::new(limits);
 let error=nepl3_wire::encode_checked(&NdfValue::Unit,&TypeDescriptor::Unit,&registry,&mut budget).unwrap_err();
 println!("actual error={error:?}; extracted={:?}; budget={:?}",error.stop_reason(),budget.poll());
 assert_eq!(budget.poll(),Err(StopReason::WorkLimit));
 let wrong=nepl3_wire::encode_checked(&NdfValue::Unit,&TypeDescriptor::Text,&registry,&mut setup).unwrap_err();
 assert_eq!(wrong.stop_reason(),None);println!("semantic control={wrong:?}");
 if error.stop_reason()!=Some(StopReason::WorkLimit){std::process::exit(2);}
}
